  const url = 'https://www.irs.gov/pub/irs-soi/' + f;
  const r = await fetch(url);
  if (!r.ok) return '### ' + f + ' HTTP ' + r.status + '\n';
  const buf = await r.arrayBuffer();
  const wb = window.__XLSX.read(new Uint8Array(buf), {type:'array'});
  let out = '### ' + f + '\n';
  for (const name of wb.SheetNames) {
    const rows = window.__XLSX.utils.sheet_to_json(wb.Sheets[name], {header:1, raw:false, defval:''});
    if (!rows.length) continue;
    const totIdx = rows.findIndex(rw => /^All returns, total/i.test(String(rw[0]).trim()));
    if (totIdx < 0) { out += 'sheet ' + name + ' no total row; first cells: ' + rows.slice(0,12).map(rw=>String(rw[0]).slice(0,40)).join(' // ') + '\n'; continue; }
    for (let i = 0; i < totIdx; i++) {
      rows[i].forEach((c, j) => {
        const s = String(c).replace(/\s+/g,' ');
        if (/\bIRA\b|individual retirement/i.test(s)) {
          const above = (rows[i-1] && rows[i-1][j]) || ''; const below = (rows[i+1] && rows[i+1][j]) || '';
          out += 'sheet ' + name + ' HDR r' + i + ' c' + j + ': [' + String(above).replace(/\s+/g,' ') + '] ' + s + ' [' + String(below).replace(/\s+/g,' ') + '] => ' + [rows[totIdx][j], rows[totIdx][j+1], rows[totIdx][j+2], rows[totIdx][j+3]].join(' | ') + '\n';
        }
      });
    }
  }
  return out;
};
