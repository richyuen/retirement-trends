window.__grabAge = async (url) => {
  const r = await fetch(url);
  if (!r.ok) return 'HTTP ' + r.status + ' ' + url + '\n';
  const buf = await r.arrayBuffer();
  const wb = window.__XLSX.read(new Uint8Array(buf), {type:'array'});
  let out = '';
  for (const name of wb.SheetNames) {
    const rows = window.__XLSX.utils.sheet_to_json(wb.Sheets[name], {header:1, raw:false, defval:''});
    if (!rows.length) continue;
    out += '### ' + url.split('/').pop() + ' SHEET ' + name + '\n';
    // find header row containing 'Withdrawals'
    let hdrIdx = rows.findIndex(rw => rw.some(c => /Withdrawals/i.test(String(c))));
    let wcol = -1, fcol = -1;
    if (hdrIdx >= 0) {
      const h = rows[hdrIdx];
      wcol = h.findIndex(c => /Withdrawals/i.test(String(c)));
      fcol = h.findIndex(c => /fair market value/i.test(String(c)) && h.indexOf(c) > wcol);
      if (fcol < 0) fcol = h.findIndex((c,i) => i > wcol && /fair market|FMV/i.test(String(c)));
    }
    out += 'TITLE: ' + String(rows[0][0]).replace(/\s+/g,' ') + '\n';
    out += 'hdrIdx=' + hdrIdx + ' wcol=' + wcol + ' fcol=' + fcol + ' hdr=' + (hdrIdx>=0 ? rows[hdrIdx].map(c=>String(c).replace(/\s+/g,' ')).filter(c=>c).join(' ; ') : '') + '\n';
    for (let i = hdrIdx + 1; i < rows.length; i++) {
      const rw = rows[i];
      const lab = String(rw[0]).replace(/\.{2,}/g,'').replace(/\s+/g,' ').trim();
      if (!lab) continue;
      if (/^(\[|\*|Note|NOTE|Source|n\.a|N\/A|n\/a|"|For additional)/.test(lab)) break;
      const vals = [rw[wcol], rw[wcol+1], rw[fcol], rw[fcol+1]].map(v => String(v === undefined ? '' : v).trim());
      if (vals.every(v => v === '')) continue;
      out += lab + '|' + vals.join('|') + '\n';
    }
  }
  return out;
};
