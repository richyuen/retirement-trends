window.__pdfPageWith = async (url, re) => {
  let doc;
  try { doc = await window.__pdfDoc(url); } catch (e) { return 'ERR ' + String(e).slice(0,120) + '\n'; }
  let out = 'URL ' + url + ' pages=' + doc.numPages + '\n';
  for (let p = 1; p <= doc.numPages; p++) {
    const page = await doc.getPage(p);
    const tc = await page.getTextContent();
    const flat = tc.items.map(i => i.str).join(' ');
    if (re.test(flat)) {
      const items = tc.items.filter(it => it.str && it.str.trim().length).map(it => ({s: it.str, x: it.transform[4], y: it.transform[5]}));
      const lines = [];
      for (const it of items) { let ln = lines.find(l => Math.abs(l.y - it.y) < 2.5); if (!ln) { ln = {y: it.y, items: []}; lines.push(ln); } ln.items.push(it); }
      lines.sort((a,b) => b.y - a.y);
      const txt = lines.map(l => l.items.sort((a,b)=>a.x-b.x).map(i=>i.s).join(' | ')).filter(l => /(\d|Mean|Median|Retired|retired|withdrawal|Number of respondents)/.test(l) && l.length < 160).join('\n');
      out += '--- p' + p + ' ---\n' + txt.replace(/[\u0000-\u0009\u000b-\u001f]+/g,'#') + '\n';
    }
  }
  return out;
};
