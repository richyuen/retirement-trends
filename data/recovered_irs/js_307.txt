window.__grab = async (url) => {
  const r = await fetch(url);
  if (!r.ok) return 'HTTP ' + r.status + ' ' + url;
  const buf = await r.arrayBuffer();
  const wb = window.__XLSX.read(new Uint8Array(buf), {type:'array'});
  let out = '';
  for (const name of wb.SheetNames) {
    const csv = window.__XLSX.utils.sheet_to_csv(wb.Sheets[name], {FS:'|', blankrows:false});
    const lines = csv.split('\n').map(l => l.replace(/\|+$/,'')).filter(l => l.replace(/\|/g,'').trim().length>0);
    out += '### SHEET ' + name + '\n' + lines.join('\n') + '\n';
  }
  return out;
};
