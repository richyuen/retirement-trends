  for (let p = 1; p <= doc.numPages; p++) {
    const page = await doc.getPage(p);
    const tc = await page.getTextContent();
    // rebuild text in reading order by lines
    const items = tc.items.filter(it => it.str && it.str.trim().length).map(it => ({s: it.str, x: it.transform[4], y: it.transform[5]}));
    const lines = [];
    for (const it of items) { let ln = lines.find(l => Math.abs(l.y - it.y) < 2.5); if (!ln) { ln = {y: it.y, items: []}; lines.push(ln); } ln.items.push(it); }
    lines.sort((a,b) => b.y - a.y);
    const txt = lines.map(l => l.items.sort((a,b)=>a.x-b.x).map(i=>i.s).join('')).join(' ').replace(/\s+/g,' ');
    const sents = txt.split(/(?<=[.;])\s+(?=[A-Z(])/);
    for (const s of sents) { if (re.test(s)) { out += 'p' + p + ': ' + s.trim().slice(0, 900) + '\n'; } }

  }
  return out;
